Examples

// A raised button: fill the shape, then draw a bevel over it, lit from the top left.
using Raster<Rgba8> button = Raster.Create(220, 90, Rgba8.White);
using (RasterCanvas<Rgba8> canvas = button.CreateCanvas())
{
    Path shape = Path.RoundRect(new RoundRect(new RectF(10, 10, 200, 70), 16));
    canvas.FillPath(shape, new Rgba8(40, 110, 200, 255));
    canvas.DrawBevel(shape, new Bevel
    {
        Profile = BevelProfile.Circle,
        Width = 10,
        Height = 8,
        Material = BevelMaterial.Plastic,
        Light = BevelLight.FromDirection(LightDirection.TopLeft),
    });
}

public BevelLight(float Angle, float Elevation, float Intensity = 1)

Parameters

Angle float

The direction the light comes from, in degrees counter-clockwise from the right as seen on the page (90 is from the top, 135 from the top left), as Photoshop's layer-effect angle. It is a device direction: rotating the canvas does not turn the light.

Elevation float

The light's height above the page, in degrees from 0 (grazing) to 90 (straight down).

Intensity float

How strong the light and shade are: 1 as the material says, 0 none, more for stronger.
