Examples

// VP8 frames in stream order, here from an IVF file (Egami.Video reads WebM and IVF for you).
using Vp8StreamDecoder decoder = new() { ConcealErrors = false };
YuvRaster<byte>? picture = null;
foreach (byte[] packet in IvfFrames("clip.ivf"))
{
    Vp8FrameInfo info = decoder.Decode(packet);
    if (!info.IsShown)
    {
        continue; // a hidden frame, typically an alternate reference: it only updates the references
    }

    picture ??= YuvRaster.Create(info.Width, info.Height, ChromaFormat.Yuv420, YuvColor.Bt601Limited);
    decoder.CopyFrameTo(picture); // the picture is the caller's: change it or keep it
    shown++;
}

picture?.Dispose();

public Vp8FrameInfo(bool IsKeyFrame, bool IsShown, int Width, int Height, int Version, bool IsConcealed)

Parameters

IsKeyFrame bool

Whether the frame is a key frame, which decodes on its own and resets the references.

IsShown bool

Whether the frame is shown; a hidden frame (typically an alternate reference) only updates the references.

Width int

The frame width, from the last key frame.

Height int

The frame height, from the last key frame.

Version int

The VP8 version, 0 to 3: 0 uses the six-tap prediction filters and the normal loop filter, 1 to 3 simpler ones.

IsConcealed bool

Whether ConcealErrors concealed broken data in this frame.
