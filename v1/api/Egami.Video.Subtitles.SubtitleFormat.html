A text subtitle format SubtitleDocument reads and writes.

public enum SubtitleFormat

Fields

Ass = 2

Advanced SubStation Alpha (.ass, script type v4.00+): styles, layers, positions and override tags.

Srt = 0

SubRip (.srt): numbered cues with HTML-style b, i, u, s and font tags.

Ssa = 3

SubStation Alpha (.ssa, script type v4.00), ASS's predecessor. Read only; written as Ass.

WebVtt = 1

WebVTT (.vtt), the W3C format browsers show: cue settings, regions, STYLE and NOTE blocks, and cue tags.

Examples

// SRT, WebVTT, ASS and SSA, told apart by their content. Saving an unchanged file gives back its exact bytes; the format
// comes from the argument, then the extension, then the format it was read as.
SubtitleDocument subtitles = SubtitleDocument.Load("movie.srt");
foreach (SubtitleCue cue in subtitles.Cues)
{
    Console.WriteLine($"{cue.Start.ToTimeSpan():g} {cue.PlainText}");
}

subtitles.Save("movie.vtt");
subtitles.Save("movie.ass");
