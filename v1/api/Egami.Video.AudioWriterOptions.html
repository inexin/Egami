Examples

// FLAC for .flac files, PCM for .wav, .w64, .mka and .mp4 unless the options name a codec; FLAC levels run 0 to 8.
using (AudioWriter writer = AudioWriter.Create("tone.flac", new AudioWriterOptions { Level = 8, BitsPerSample = 24 }))
{
    writer.Write(tone);
}

// 16-bit WAV at 44.1 kHz: the writer resamples and converts.
using (AudioWriter writer = AudioWriter.Create("tone.wav", new AudioWriterOptions { SampleRate = 44100, BitsPerSample = 16 }))
{
    writer.Write(tone);
}

Gets the bits per sample stored, or 0 to follow the first frame: its valid bits for integer samples (16 for 16-bit, 24 for 24-bit in S32), and for float samples float PCM of the same size, 24-bit FLAC or ALAC, or 16-bit integer PCM.

Examples

// Speech at 32 kbit/s. Opus codes 8, 12, 16, 24 and 48 kHz, so this 44.1 kHz audio is resampled to 48 kHz; the
// header keeps 44.1 kHz as the rate the audio came at.
using (AudioWriter writer = AudioWriter.Create("voice.opus", new AudioWriterOptions
{
    Opus = new OpusEncoderOptions { Bitrate = 32_000, Application = OpusApplication.Voice },
}))
{
    writer.Write(tone);
}
