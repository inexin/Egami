public enum BarcodeFormat

Fields

Aztec = 3

Aztec Code (ISO/IEC 24778); no quiet zone, as its finder is in the middle.

Code128 = 4

Code 128 and GS1-128 (ISO/IEC 15417), linear; quiet zone 10 modules.

Code39 = 5

Code 39 (ISO/IEC 16388), linear; quiet zone 10 modules.

DataMatrix = 1

Data Matrix ECC 200 (ISO/IEC 16022); quiet zone 1 module.

Ean13 = 7

EAN-13 (ISO/IEC 15420), linear; quiet zone 11 modules.

Ean8 = 8

EAN-8 (ISO/IEC 15420), linear; quiet zone 7 modules.

Itf = 6

Interleaved 2 of 5 and ITF-14 (ISO/IEC 16390), linear; quiet zone 10 modules.

Pdf417 = 2

PDF417 (ISO/IEC 15438); quiet zone 2 modules.

Qr = 0

QR Code (ISO/IEC 18004); quiet zone 4 modules.

UpcA = 9

UPC-A (ISO/IEC 15420), linear; quiet zone 9 modules.

UpcE = 10

UPC-E (ISO/IEC 15420), linear; quiet zone 9 modules.

Examples

// Modules from another encoder become a barcode with the quiet zone of their standard: 4 modules for QR.
Barcode qr = Barcode.FromModules(BarcodeFormat.Qr, modules);
Console.WriteLine($"{qr.Format}: {qr.Modules.Width} modules plus {qr.QuietZone} on each side");
