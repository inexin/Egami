Enum GraphicsUnit

Namespace
Egami.SystemDrawing
Assembly

The unit of a measure: the world's, the display's, pixels, points, inches, document units or millimeters.

public enum GraphicsUnit

Fields

Display = 1

The display's unit: pixels on screens, 1/100 inch on printers.

Document = 5

A document unit, 1/300 inch.

Inch = 4

An inch.

Millimeter = 6

A millimeter.

Pixel = 2

A device pixel.

Point = 3

A printer's point, 1/72 inch.

World = 0

The world coordinate system's unit.

Examples

// The same code as with System.Drawing.Common: only the using lines name Egami.SystemDrawing.
using Bitmap bitmap = new(64, 32);
for (int x = 0; x < 64; x++)
{
    bitmap.SetPixel(x, 16, Color.FromArgb(255, x * 4, 0, 255 - (x * 4)));
}

Color middle = bitmap.GetPixel(32, 16);  // ARGB(255, 128, 0, 127)
bitmap.SetResolution(300, 300);
bitmap.RotateFlip(RotateFlipType.Rotate90FlipNone);
bitmap.Save(output, ImageFormat.Png);

GraphicsUnit unit = GraphicsUnit.Inch;
RectangleF bounds = bitmap.GetBounds(ref unit);  // 32 by 64, in pixels
