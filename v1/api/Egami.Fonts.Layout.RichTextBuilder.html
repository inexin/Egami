Class RichTextBuilder

Namespace
Egami.Fonts.Layout
Assembly
Egami.Fonts.dll

Builds a RichText: Push(TextStyle) a style, Append(string) text, Pop() back to the enclosing style. Styles nest: a pushed style's unset fields come from the one it is pushed on, and the result has one flat span per stretch of equal style.

Examples

// Font families come from a collection; later families cover what the first lacks.
FontCollection fonts = new();
fonts.AddFile(System.IO.Path.Combine(folder, "NotoSans-Regular.ttf"));
fonts.AddFile(System.IO.Path.Combine(folder, "NotoSansArabic-Regular.ttf"));
fonts.TryGetFamily("Noto Sans", out FontFamily latin);
fonts.TryGetFamily("Noto Sans Arabic", out FontFamily arabic);

// Styles nest: the inner span keeps the outer one's size and adds a color and an underline.
RichText text = new RichTextBuilder()
    .Append("Peace is ")
    .Push(new TextStyle { Size = 20 })
    .Append("\u0633\u0644\u0627\u0645 ")
    .Push(new TextStyle { Color = new Rgba8(200, 30, 30, 255), Decoration = new TextDecoration() })
    .Append("in Arabic")
    .Pop()
    .Pop()
    .AddPlaceholder(new Placeholder(16, 16, PlaceholderAlignment.Middle))
    .ToRichText();

ParagraphStyle style = new()
{
    DefaultStyle = new TextStyle { FontFamilies = [latin, arabic], Size = 16, LineHeight = LineHeight.Multiple(1.4f) },
    Align = TextAlign.Justify,
    LineBreaking = new LineBreakOptions(WordBreak: WordBreak.Normal),
};

// Shaped once: lay it out at any width later.
Paragraph paragraph = Paragraph.Create(text, style);

Examples

// 国際 (kokusai) with its reading as group ruby, and 価値 (kachi) as jukugo ruby: one annotation per character.
RichText text = new RichTextBuilder()
    .Append("国際の価値")
    .AddRuby(0, 2, "こくさい")
    .AddRuby(3, 1, "か")
    .AddRuby(4, 1, "ち")
    .ToRichText();

ParagraphStyle style = new()
{
    DefaultStyle = new TextStyle { FontFamilies = [fonts.Families[0]], Size = 20, Language = "ja" },
    Ruby = new RubyOptions { Merge = RubyMerge.Auto },  // jukugo: per character, joined when a reading is too wide
};
TextLayout layout = Paragraph.Create(text, style).Layout(300);

// The ruby text: runs beside the line's text, at half the base's size, and the box each annotation took.
foreach (LayoutRun run in layout.Lines[0].AnnotationRuns)
{
    TextBounds box = layout.GetRubyBounds(run.Annotation);
}

// An annotation of its own: under the base, in red, a little away from it.
RubyAnnotation under = new(0, 2, "kokusai")
{
    Position = RubyPosition.Under,
    Offset = 2,
    Style = new TextStyle { Color = new Rgba8(200, 0, 0, 255) },
};
