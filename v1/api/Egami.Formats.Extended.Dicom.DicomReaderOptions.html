Options for reading DICOM: which frame, what values, and which VOI transform shows a monochrome image. Pass them to Load(Stream, EgamiContext?, ReaderOptions?) or LoadFrame(int, DicomReaderOptions?, EgamiContext?).

[Experimental("EGAMIX001", UrlFormat = "https://github.com/inexin/Egami/blob/main/docs/articles/formats/dicom.md")]
public sealed record DicomReaderOptions : ReaderOptions, IEquatable<ReaderOptions>, IEquatable<DicomReaderOptions>

Examples

// A window of your own, in modality values (Hounsfield units for CT), or the file's second one.
using Raster bone = Raster.Load("scan.dcm", context, new DicomReaderOptions { WindowCenter = 400, WindowWidth = 1800 });
using Raster second = Raster.Load("scan.dcm", context, new DicomReaderOptions { VoiIndex = 1 });
using Raster smooth = Raster.Load("scan.dcm", context,
    new DicomReaderOptions { WindowCenter = 40, WindowWidth = 400, WindowFunction = DicomVoiFunction.Sigmoid });
