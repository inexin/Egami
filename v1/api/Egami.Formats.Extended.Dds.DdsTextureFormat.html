Egami.Formats.Extended.Dds

The formats the DDS writer writes, each a DXGI format; Srgb picks the _SRGB form where there is one.

public enum DdsTextureFormat

Fields

A8 = 5

A8_UNORM: alpha only. Legacy header A8.

Bc1 = 19

BC1 (DXT1): RGB with 1-bit alpha, 4 bits a pixel.

Bc2 = 20

BC2 (DXT3): RGB with explicit 4-bit alpha, 8 bits a pixel.

Bc3 = 21

BC3 (DXT5): RGB with interpolated alpha, 8 bits a pixel.

Bc4 = 22

BC4 (ATI1): one channel, the raster's gray, 4 bits a pixel.

Bc5 = 23

BC5 (ATI2): red and green, for normal maps, 8 bits a pixel.

Bc6h = 25

BC6H_UF16: unsigned half-float RGB for HDR, 8 bits a pixel; negatives become 0. DX10 header.

Bc6hSigned = 26

BC6H_SF16: signed half-float RGB for HDR, 8 bits a pixel. DX10 header.

Bc7 = 24

BC7: RGBA at 8 bits a pixel, with eight modes and partitions; the best quality of the 8-bit block formats. DX10 header.

Bgr565 = 16

B5G6R5_UNORM: 16-bit colour. Legacy header R5G6B5.

Bgra4444 = 18

B4G4R4A4_UNORM: 4 bits a channel. Legacy header A4R4G4B4.

Bgra5551 = 17

B5G5R5A1_UNORM: 15-bit colour and 1-bit alpha. Legacy header A1R5G5B5.

Bgra8 = 1

B8G8R8A8_UNORM: 8-bit BGRA. Legacy header A8R8G8B8.

Bgrx8 = 2

B8G8R8X8_UNORM: 8-bit BGR with an unused byte. Legacy header X8R8G8B8.

R11G11B10F = 14

R11G11B10_FLOAT: unsigned 11-, 11- and 10-bit floats, for HDR colour without alpha.

R16 = 6

R16_UNORM: one 16-bit channel, the raster's gray. Legacy header L16.

R32F = 10

R32_FLOAT: one float channel, the raster's gray.

R8 = 3

R8_UNORM: one 8-bit channel, the raster's gray. Legacy header L8.

Rg16 = 7

R16G16_UNORM: red and green, 16 bits each. Legacy header G16R16.

Rg8 = 4

R8G8_UNORM: red and green, 8 bits each, as two-channel normal maps keep them.

Rgb10A2 = 13

R10G10B10A2_UNORM: 10-bit RGB and 2-bit alpha.

Rgb9E5 = 15

R9G9B9E5_SHAREDEXP: three 9-bit mantissas sharing an exponent, for HDR colour without alpha.

RgbF32 = 11

R32G32B32_FLOAT: float RGB.

Rgba16 = 8

R16G16B16A16_UNORM: 16-bit RGBA. Legacy header D3DFMT 36.

Rgba8 = 0

R8G8B8A8_UNORM: 8-bit RGBA. Legacy header A8B8G8R8.

RgbaF16 = 9

R16G16B16A16_FLOAT: half-float RGBA.

RgbaF32 = 12

R32G32B32A32_FLOAT: float RGBA.

Examples

// A BC3 texture with its full mip chain: the format, mip count, effort (0 to 6) and header are options.
using (FileStream output = File.Create("duck.dds"))
{
    image.Save(output, DdsFormat.Info, new DdsWriterOptions { Format = DdsTextureFormat.Bc3, Effort = 5 });
}

// An sRGB colour texture without mips, in the DX10 header.
using (FileStream output = File.Create("duck-srgb.dds"))
{
    image.Save(output, DdsFormat.Info, new DdsWriterOptions { Format = DdsTextureFormat.Bc1, Srgb = true, MipLevels = 1 });
}

// Six faces back into a cube map; SaveAll without Cube writes a texture array.
using RasterList faces = Raster.LoadAll("sky.dds", context);
using (FileStream output = File.Create("sky-bc1.dds"))
{
    Raster.SaveAll(faces, output, DdsFormat.Info, new DdsWriterOptions { Cube = true, Format = DdsTextureFormat.Bc1 });
}
