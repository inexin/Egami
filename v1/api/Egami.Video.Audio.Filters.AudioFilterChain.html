Class AudioFilterChain

Namespace
Egami.Video.Audio.Filters
Assembly
Egami.Video.dll

Runs several filters on one input in a single pass: each frame is converted to F32Planar once, then handed from filter to filter in order, in place where the filter keeps the length. The chain is itself an IAudioFilter, owns the filters added to it and disposes them with itself.

public sealed class AudioFilterChain : AudioFilter, IAudioFilter, IDisposable

Examples

// Fade in over 2 s, then turn down 3 dB: frames of any format go in, F32 planar frames come out.
using AudioFilterChain chain = new AudioFilterChain(48000, ChannelLayout.Stereo)
    .Add(new Fade(FadeDirection.In, MediaTime.Zero, MediaTime.FromSeconds(2)))
    .Add(new Gain(-3));
foreach (AudioFrame frame in frames)
{
    chain.Send(frame);
    while (chain.TryReceive(out AudioFrame? output))
    {
        samples += output.SampleCount; // write it to an AudioWriter or encoder here
        output.Dispose();
    }
}

chain.Flush();
