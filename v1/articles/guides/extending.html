Custom operations and formats

Everything here uses only Egami's public API; Egami's own test suite has an outside project that does each of these things.

Operations

An operation implements IRasterOperation: it says what size it makes and fills the destination rows.

/// <summary>Swaps red and blue, in parallel bands of rows.</summary>
public sealed class SwapRedBlue : IRasterOperation
{
    public Size OutputSize(Size input) => input;

    public void Run<TPixel>(OperationInput<TPixel> input)
        where TPixel : unmanaged, IPixelType<TPixel>
    {
        input.ForEachBand((start, end) =>
        {
            for (int y = start; y < end; y++)
            {
                Span<TPixel> source = input.Source.Row(y);
                Span<TPixel> destination = input.Destination.Row(y);
                for (int x = 0; x < source.Length; x++)
                {
                    Vector4 v = source[x].ToScaled();
                    destination[x] = TPixel.FromScaled(new Vector4(v.Z, v.Y, v.X, v.W));
                }
            }
        });
    }
}

image.Apply(ops => ops.Fit(100, 100).Add(new SwapRedBlue()), TestContext.Current.CancellationToken);
  • OutputSize is asked while the chain is planned, before any pixel changes. Throw there for arguments that cannot work.
  • When the size stays the same, Source and Destination may be the same plane (IsInPlace); read a row before you write it.
  • ForEachBand runs bands of rows on up to EgamiContext.MaxParallelism threads and checks the cancellation token between bands.
  • A step that changes the size updates the EXIF dimensions and drops the thumbnail, as the built-in resizes do.
  • TPixel.Layout tells you the pixel type, so you can take a fast path for the types you know and fall back to ToScaled and FromScaled for the rest. Check Layout.Alpha before you assume straight alpha.

To run a chain on part of a larger picture, such as a tile or a canvas layer, pad the part by how far the chain reads: RasterOps.Reach returns that as an OperationReach, or null when a step reads the whole image (a resize, a turn, a custom operation).

// How far a chain reads around each pixel: run it on a part of a picture padded by this much, and the part comes out as it
// would in the whole picture.
OperationReach? reach = RasterOps.Reach(ops => ops.Blur(3).Glow(4, Rgba8.White));
if (reach is OperationReach r)
{
    Console.WriteLine($"pad by {r.X} × {r.Y}");
}

// A resize reads the whole image: its reach is unknown.
Assert.Null(RasterOps.Reach(ops => ops.Resize(100, 100)));

Formats

A format is a FormatInfo: a name, extensions, a signature check, a reader and a writer.

/// <summary>A minimal format: "RAW8", width and height as big-endian numbers, then RGBA rows.</summary>
public static class Raw8Format
{
    public static FormatInfo Info { get; } = new()
    {
        Name = "RAW8",
        Extensions = ["raw8"],
        SignatureLength = 4,
        Matches = header => header.StartsWith("RAW8"u8),
        Reader = new Reader(),
        Writer = new Writer(),
    };

    private sealed class Reader : IFormatReader
    {
        public RasterInfo ReadInfo(EgamiInput input, ReaderOptions options)
        {
            (int width, int height) = Header(input);
            return new RasterInfo { Format = Info, Width = width, Height = height, Layout = Rgba8.Layout, FrameCount = 1, EstimatedBytes = 4L * width * height };
        }

        public Raster<TPixel> Read<TPixel>(EgamiInput input, ReaderOptions options)
            where TPixel : unmanaged, IPixelType<TPixel>
        {
            (int width, int height) = Header(input);

            // CreateRaster checks the size against the safety limits before allocating.
            Raster<TPixel> raster = input.CreateRaster<TPixel>(width, height);
            using IMemoryOwner<byte> row = input.RentScratch(width * 4);
            for (int y = 0; y < height; y++)
            {
                input.ReadExactly(row.Memory.Span[..(width * 4)]);
                PixelConvert.Convert<Rgba8, TPixel>(MemoryMarshal.Cast<byte, Rgba8>(row.Memory.Span[..(width * 4)]), raster.Row(y));
            }

            return raster;
        }

        public Raster Read(EgamiInput input, ReaderOptions options) => Read<Rgba8>(input, options);

        private static (int Width, int Height) Header(EgamiInput input)
        {
            input.Skip(4);
            return ((int)input.ReadUInt32BigEndian(), (int)input.ReadUInt32BigEndian());
        }
    }

    private sealed class Writer : IFormatWriter
    {
        public void Write<TPixel>(Raster<TPixel> raster, Stream output, WriterOptions options)
            where TPixel : unmanaged, IPixelType<TPixel>
        {
            Span<byte> header = stackalloc byte[12];
            "RAW8"u8.CopyTo(header);
            BinaryPrimitives.WriteUInt32BigEndian(header[4..], (uint)raster.Width);
            BinaryPrimitives.WriteUInt32BigEndian(header[8..], (uint)raster.Height);
            output.Write(header);
            Rgba8[] row = new Rgba8[raster.Width];
            for (int y = 0; y < raster.Height; y++)
            {
                PixelConvert.Convert<TPixel, Rgba8>(raster.Row(y), row);
                output.Write(MemoryMarshal.AsBytes(row.AsSpan()));
            }
        }
    }
}

// Register it in a context...
EgamiContext context = EgamiContext.Default.WithFormat(Raw8Format.Info);
using (FileStream output = File.Create("duck.raw8"))
{
    image.Save(output, Raw8Format.Info);
}

using Raster back = Raster.Load("duck.raw8", context);

// ...or, once at startup before any other Egami call, make it part of the default context:
// EgamiContext.SetDefault(new EgamiContext().WithFormat(Raw8Format.Info));

Readers read only through EgamiInput, which keeps them inside the safety limits:

Member Use
Peek, Read, ReadExactly, Skip, ReadByte, ReadUInt16/32… Read the data; a short read throws CorruptDataException
CreateRaster<TPixel>(width, height) The raster you decode into, size-checked before any allocation
AddFrame(raster) Another frame for animated formats, counted against MaxFrames and the memory budget
RentScratch(bytes) Working memory counted against the same budget
CheckMemory, CheckFrameCount, CheckMetadataBytes Check a number from the header before acting on it
Warn(raster, id, text) Note something a lenient read recovered from, with a stable id such as Pcx.PaletteMissing
Corrupt(offset, problem) The exception to throw for data that breaks the format's rules

Respect ReaderOptions.Recovery: in Lenient mode keep what decoded and warn; in Strict mode throw. The writer gets the raster and a stream; options come as your own WriterOptions subclass.

Streaming

Two optional interfaces let a format stream. A reader that also implements IRowFormatReader returns a RowReader<TNatural> from Open: the open image hands out rows as RasterDecoder asks for them (top-down, never going back within a frame) and moves through frames with NextFrame. The decoder applies Region, MinimumSize and the pixel conversion after it. A writer that also implements IRowFormatWriter returns a RowEncoder<TPixel> from Start, which takes rows (or frames, for an animation) one at a time; Save, SaveAsync, RasterEncoder and Transcode then all write through it. When streaming, EncodeStart.Whole is null: an option that needs every pixel before the first byte should call EncodeStart.Refuse.

/// <summary>The same RAW8 format, streaming: rows go in and out one at a time, so RasterDecoder, RasterEncoder and Transcode stream it.</summary>
public static class StreamingRaw8Format
{
    public static FormatInfo Info { get; } = Raw8Format.Info with { Reader = new Reader(), Writer = new Writer() };

    private sealed class Reader : IFormatReader, IRowFormatReader
    {
        private static readonly IFormatReader Whole = Raw8Format.Info.Reader!;

        public RasterInfo ReadInfo(EgamiInput input, ReaderOptions options) => Whole.ReadInfo(input, options) with { Access = RasterAccess.Streamed };

        public Raster<TPixel> Read<TPixel>(EgamiInput input, ReaderOptions options)
            where TPixel : unmanaged, IPixelType<TPixel> => Whole.Read<TPixel>(input, options);

        public Raster Read(EgamiInput input, ReaderOptions options) => Whole.Read(input, options);

        // The open image reads each row when it is asked for; the decoder crops, reduces and converts after it.
        public RowReader Open(EgamiInput input, ReaderOptions options)
        {
            input.Skip(4);
            int width = (int)input.ReadUInt32BigEndian();
            return new Rows(input, width, (int)input.ReadUInt32BigEndian());
        }
    }

    private sealed class Rows(EgamiInput input, int width, int height) : RowReader<Rgba8>
    {
        private readonly Rgba8[] _row = new Rgba8[width];
        private int _next;
        private bool _started;

        public override int Width => width;

        public override int Height => height;

        public override Metadata.RasterMetadata Metadata { get; } = new();

        public override Metadata.FrameMetadata? NextFrame()
        {
            if (_started)
            {
                return null;
            }

            _started = true;
            return new Metadata.FrameMetadata();
        }

        public override void ReadRow<TPixel>(int y, int x, Span<TPixel> destination)
        {
            for (; _next <= y; _next++)
            {
                input.ReadExactly(MemoryMarshal.AsBytes(_row.AsSpan()));
            }

            PixelConvert.Convert<Rgba8, TPixel>(_row.AsSpan(x, destination.Length), destination);
        }
    }

    private sealed class Writer : IFormatWriter, IRowFormatWriter
    {
        public void Write<TPixel>(Raster<TPixel> raster, Stream output, WriterOptions options)
            where TPixel : unmanaged, IPixelType<TPixel> => Raw8Format.Info.Writer!.Write(raster, output, options);

        // Save, SaveAsync, RasterEncoder and Transcode all write through the encoder.
        public RowEncoder<TPixel> Start<TPixel>(Stream output, EncodeStart<TPixel> start)
            where TPixel : unmanaged, IPixelType<TPixel> => new Encoder<TPixel>(output, start.Width, start.Height);
    }

    private sealed class Encoder<TPixel> : RowEncoder<TPixel>
        where TPixel : unmanaged, IPixelType<TPixel>
    {
        private readonly Stream _output;
        private readonly Rgba8[] _row;

        public Encoder(Stream output, int width, int height)
        {
            _output = output;
            _row = new Rgba8[width];
            Span<byte> header = stackalloc byte[12];
            "RAW8"u8.CopyTo(header);
            BinaryPrimitives.WriteUInt32BigEndian(header[4..], (uint)width);
            BinaryPrimitives.WriteUInt32BigEndian(header[8..], (uint)height);
            output.Write(header);
        }

        public override void WriteRow(ReadOnlySpan<TPixel> row)
        {
            PixelConvert.Convert<TPixel, Rgba8>(row, _row);
            _output.Write(MemoryMarshal.AsBytes(_row.AsSpan()));
        }

        public override void Complete()
        {
        }
    }
}

A format without them still works everywhere: the decoder reads the whole image with Read and serves rows from it, and the encoder gathers the image and calls Write at the end.

Pixel types

A pixel type is an unmanaged struct implementing IPixelType<TSelf>. Its Layout says how it is stored: a layout Egami knows (Rgba with Unorm8, say) lets Egami take its fast paths; anything else uses PixelChannelOrder.Other, and Egami converts it only through its own ToScaled and FromScaled.

Wrapping memory

Raster.Wrap puts a raster over memory you own, without copying: a Memory<TPixel> with a stride in pixels, or a Memory<byte> with a stride in bytes, such as a video frame whose rows are padded. Pass an IDisposable owner to have the raster release the memory when it is disposed.

Rows and planes

For a one-off pass over pixels, ForEachRow runs a lambda on each row in parallel; no operation class needed.

using Raster<Rgba8> image = Raster.Create<Rgba8>(256, 64);

// A horizontal gradient; rows run in parallel, each row is a Span over the pixels.
image.ForEachRow((row, y) =>
{
    for (int x = 0; x < row.Length; x++)
    {
        row[x] = new Rgba8((byte)x, (byte)(y * 4), 128, 255);
    }
});

Each frame's pixels are a PixelPlane<TPixel>: rows of Span<TPixel>, and the whole frame as one span when it is stored in one block.

PixelPlane<Rgba8> plane = image.Frames[0].Plane;
if (plane.TryGetContiguous(out Span<Rgba8> all))
{
    all.Fill(Rgba8.White);  // one Span over every pixel when the rows are stored back to back
}
else
{
    for (int y = 0; y < plane.Height; y++)
    {
        plane.Row(y).Fill(Rgba8.White);
    }
}

RasterOps records each step of a chain; built-in steps and your own mix freely.

// RasterOps records steps; Apply runs them in one pass where it can, in place when the size stays.
using Raster photo = Raster.Load("egret.png");
photo.Apply(ops => ops
    .AutoOrient()
    .Cover(400, 300, Anchor.Center)
    .Saturation(1.1f)
    .Sharpen(1, 0.6f));

/// <summary>Black or white per pixel: ForEachBand gives each worker a band of rows from start to end.</summary>
private sealed class Threshold(float level) : IRasterOperation
{
    public Size OutputSize(Size input) => input;

    public void Run<TPixel>(OperationInput<TPixel> input)
        where TPixel : unmanaged, IPixelType<TPixel>
    {
        RowBandAction band = (start, end) =>
        {
            for (int y = start; y < end; y++)
            {
                Span<TPixel> source = input.Source.Row(y), destination = input.Destination.Row(y);
                for (int x = 0; x < source.Length; x++)
                {
                    Vector4 v = source[x].ToScaled();
                    float on = ((0.2126f * v.X) + (0.7152f * v.Y) + (0.0722f * v.Z)) >= level ? 1 : 0;
                    destination[x] = TPixel.FromScaled(new Vector4(on, on, on, v.W));
                }
            }
        };
        input.ForEachBand(band);
    }
}

Recipes

Short answers to common tasks in this area. Each one compiles and runs in Egami's test run.

Interop with raw buffers

// Pixels from somewhere else, such as a camera SDK: 640 × 480 BGRA, rows 2,560 bytes apart.
byte[] buffer = new byte[2560 * 480];

// Wrap them without copying, work on them, and save.
using (Raster<Bgra8> frame = Raster.Wrap<Bgra8>(buffer, 640, 480, byteStride: 2560))
{
    frame.Apply(ops => ops.FillColor(Rgba8.Black).Opacity(1));
    frame.Save("frame.png");
}

// Or copy the pixels out of a raster into your own buffer, row by row.
using Raster<Rgba8> image = Raster.Load<Rgba8>("duck.png");
Rgba8[] pixels = new Rgba8[image.Width * image.Height];
for (int y = 0; y < image.Height; y++)
{
    image.Row(y).CopyTo(pixels.AsSpan(y * image.Width, image.Width));
}
